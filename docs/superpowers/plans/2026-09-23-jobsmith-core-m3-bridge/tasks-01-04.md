```typescript
import type { ArtifactOrigin } from "./values";

export const ARTIFACT_KINDS = [
  { kind: "research", tab: "research", label: "Research", sendable: false, companyWide: true },
  { kind: "fit_brief", tab: "research", label: "Fit brief", sendable: false, companyWide: true },
  { kind: "people_notes", tab: "research", label: "People notes", sendable: false, companyWide: true },
  { kind: "cv", tab: "documents", label: "CV", sendable: true, companyWide: false },
  { kind: "cover_letter", tab: "documents", label: "Cover letter", sendable: true, companyWide: false },
  { kind: "message_draft", tab: "documents", label: "Message", sendable: true, companyWide: false },
  { kind: "question_bank", tab: "prep", label: "Question bank", sendable: false, companyWide: false },
  { kind: "call_card", tab: "prep", label: "Call card", sendable: false, companyWide: false },
  { kind: "pitch", tab: "prep", label: "Pitch", sendable: false, companyWide: false },
  { kind: "glossary", tab: "prep", label: "Glossary", sendable: false, companyWide: false },
  { kind: "debrief", tab: "prep", label: "Debrief", sendable: false, companyWide: false },
  { kind: "other", tab: "prep", label: "Other", sendable: false, companyWide: false },
] as const satisfies readonly { kind: string; tab: "research" | "documents" | "prep"; label: string; sendable: boolean; companyWide: boolean }[];
export type ArtifactKind = (typeof ARTIFACT_KINDS)[number]["kind"];
export type ArtifactTab = "research" | "documents" | "prep";
export const ARTIFACT_KIND_VALUES = ARTIFACT_KINDS.map((k) => k.kind) as [ArtifactKind, ...ArtifactKind[]];

export function kindInfo(kind: ArtifactKind): (typeof ARTIFACT_KINDS)[number] {
  // ARTIFACT_KINDS covers every ArtifactKind by construction (ARTIFACT_KIND_VALUES
  // is derived from it), so this always finds a match for a value that
  // type-checks as ArtifactKind.
  return ARTIFACT_KINDS.find((k) => k.kind === kind)!;
}

export function parseKind(value: string): ArtifactKind | null {
  const normalized = value.trim().toLowerCase();
  const match = ARTIFACT_KINDS.find((k) => k.kind === normalized);
  return match ? match.kind : null;
}

export function kindsForTab(tab: ArtifactTab): ArtifactKind[] {
  return ARTIFACT_KINDS.filter((k) => k.tab === tab).map((k) => k.kind);
}

export const DEFAULT_KIND_FOR_TAB: Record<ArtifactTab, ArtifactKind> = {
  research: "research",
  documents: "cv",
  prep: "question_bank",
};

export const ORIGIN_WORDS: Record<ArtifactOrigin, string> = {
  pushed: "Pushed",
  pasted: "Pasted",
  manual: "Written in the app",
  generated: "Generated",
};
```

The `ARTIFACT_KIND_VALUES` cast to a non-empty tuple follows the same reasoning `lib/pipeline/values.ts`'s `STAGE_KIND_VALUES` already uses in this repo: `Array.prototype.map`'s library signature returns a plain array, not a tuple, but Drizzle's `text(name, { enum })` (Step 6 below) requires a non-empty tuple; the array is a compile-time-only concern here since `ARTIFACT_KINDS` itself is a literal `as const` array of 12 known entries.

- [ ] **Step 6: Write `lib/db/schema/artifact.ts` in full**

D2's tenant-safety rules translated into Drizzle: composite foreign keys `(user_id, opportunity_id)` and `(user_id, company_id)`, both cascade; a null scope column skips its own FK (Postgres `MATCH SIMPLE`, the default for a multi-column FK, matches when any column is null); `artifact_scope_check` enforces exactly one scope in SQL; `stage_id` is a single-column FK with `SET NULL`, the same shape `event.stage_id` already uses in `lib/db/schema/pipeline.ts`; `artifact_company_stage_check` forbids a stage on a company-scoped row. `unique(user_id, id)` is added now (D2) even though nothing in this milestone references an `artifact` row as a parent, because a later cycle's `prep_question` is expected to.

```typescript
import { pgTable, text, uuid, integer, timestamp, foreignKey, unique, uniqueIndex, check } from "drizzle-orm/pg-core";
import { sql } from "drizzle-orm";
import { user } from "./auth";
import { company, opportunity, stage } from "./pipeline";
import { ARTIFACT_KIND_VALUES } from "@/lib/artifacts/kinds";
import { ARTIFACT_ORIGINS } from "@/lib/artifacts/values";

export const artifact = pgTable(
  "artifact",
  {
    id: uuid("id").primaryKey().defaultRandom(),
    userId: text("user_id").notNull().references(() => user.id, { onDelete: "cascade" }),
    opportunityId: uuid("opportunity_id"),
    companyId: uuid("company_id"),
    stageId: uuid("stage_id").references(() => stage.id, { onDelete: "set null" }),
    key: text("key").notNull(),
    version: integer("version").notNull(),
    kind: text("kind", { enum: ARTIFACT_KIND_VALUES }).notNull(),
    title: text("title").notNull(),
    bodyMd: text("body_md").notNull(),
    contentHash: text("content_hash").notNull(),
    sourceHash: text("source_hash"),
    origin: text("origin", { enum: ARTIFACT_ORIGINS }).notNull(),
    editedAt: timestamp("edited_at", { withTimezone: true }),
    sentAt: timestamp("sent_at", { withTimezone: true }),
    createdAt: timestamp("created_at", { withTimezone: true }).defaultNow().notNull(),
    updatedAt: timestamp("updated_at", { withTimezone: true }).defaultNow().notNull(),
  },
  (t) => [
    unique("artifact_user_id_id_unique").on(t.userId, t.id),
    uniqueIndex("artifact_opportunity_key_version_unique")
      .on(t.opportunityId, t.key, t.version)
      .where(sql`${t.opportunityId} is not null`),
    uniqueIndex("artifact_company_key_version_unique")
      .on(t.companyId, t.key, t.version)
      .where(sql`${t.companyId} is not null`),
    foreignKey({
      name: "artifact_opportunity_fk",
      columns: [t.userId, t.opportunityId],
      foreignColumns: [opportunity.userId, opportunity.id],
    }).onDelete("cascade"),
    foreignKey({
      name: "artifact_company_fk",
      columns: [t.userId, t.companyId],
      foreignColumns: [company.userId, company.id],
    }).onDelete("cascade"),
    check("artifact_scope_check", sql`(${t.opportunityId} is null) <> (${t.companyId} is null)`),
    check("artifact_company_stage_check", sql`${t.companyId} is null or ${t.stageId} is null`),
    check("artifact_version_check", sql`${t.version} >= 1`),
    check(
      "artifact_kind_check",
      sql`${t.kind} in ('research', 'fit_brief', 'people_notes', 'cv', 'cover_letter', 'message_draft', 'question_bank', 'call_card', 'pitch', 'glossary', 'debrief', 'other')`,
    ),
    check("artifact_origin_check", sql`${t.origin} in ('pushed', 'pasted', 'manual', 'generated')`),
  ],
);
```

- [ ] **Step 4: Write `lib/db/scoped/artifact.ts` in full**

`listLatestForOpportunity`/`listLatestForCompany` are the `selectDistinctOn` case D20 calls out; `listVersions`/`getVersion`/`getLatest`/`listKeys` share one small `scopeCondition` helper that turns an `ArtifactScopeRef` into a single `eq(...)` on whichever id is present. No extra "and the other id is null" clause is needed: `artifact_scope_check` (Task 1) guarantees a company-scoped row's `opportunity_id` is always null and vice versa, so filtering on one id can never accidentally match a row from the other scope. Verified in a planning scratch file: a company-scoped and an opportunity-scoped row sharing the same `key` never leak into each other's `scopeCondition` result.

```typescript
import { and, asc, desc, eq, getTableColumns } from "drizzle-orm";
import * as schema from "../schema";
import type { Db } from "../client";
import { stripScopedKeys } from "./strip";

export type ArtifactRow = typeof schema.artifact.$inferSelect;
export type ArtifactFields = Omit<typeof schema.artifact.$inferInsert, "id" | "userId" | "createdAt" | "updatedAt">;
export type ArtifactMeta = Omit<ArtifactRow, "bodyMd" | "userId">;
export type ArtifactScopeRef = { opportunityId: string } | { companyId: string };

function scopeCondition(ref: ArtifactScopeRef) {
  return "opportunityId" in ref
    ? eq(schema.artifact.opportunityId, ref.opportunityId)
    : eq(schema.artifact.companyId, ref.companyId);
}

export function artifactQueries(db: Db, userId: string) {
  // getTableColumns(...) returns a plain column map; destructuring off two
  // keys is enough to turn it into a valid `.select()` argument that leaves
  // out bodyMd and userId from every list-shaped read below.
  const { bodyMd: _bodyMd, userId: _userId, ...metaColumns } = getTableColumns(schema.artifact);
  void _bodyMd;
  void _userId;

  return {
    async listLatestForOpportunity(opportunityId: string): Promise<ArtifactMeta[]> {
      return db
        .selectDistinctOn([schema.artifact.key], metaColumns)
        .from(schema.artifact)
        .where(and(eq(schema.artifact.userId, userId), eq(schema.artifact.opportunityId, opportunityId)))
        .orderBy(schema.artifact.key, desc(schema.artifact.version));
    },
    async listLatestForCompany(companyId: string): Promise<ArtifactMeta[]> {
      return db
        .selectDistinctOn([schema.artifact.key], metaColumns)
        .from(schema.artifact)
        .where(and(eq(schema.artifact.userId, userId), eq(schema.artifact.companyId, companyId)))
        .orderBy(schema.artifact.key, desc(schema.artifact.version));
    },
    async listVersions(ref: ArtifactScopeRef, key: string): Promise<ArtifactMeta[]> {
      return db
        .select(metaColumns)
        .from(schema.artifact)
        .where(and(eq(schema.artifact.userId, userId), scopeCondition(ref), eq(schema.artifact.key, key)))
        .orderBy(asc(schema.artifact.version));
    },
    async getVersion(ref: ArtifactScopeRef, key: string, version: number): Promise<ArtifactRow | null> {
      const [row] = await db
        .select()
        .from(schema.artifact)
        .where(
          and(
            eq(schema.artifact.userId, userId),
            scopeCondition(ref),
            eq(schema.artifact.key, key),
            eq(schema.artifact.version, version),
          ),
        );
      return row ?? null;
    },
    async getLatest(ref: ArtifactScopeRef, key: string): Promise<ArtifactRow | null> {
      const [row] = await db
        .select()
        .from(schema.artifact)
        .where(and(eq(schema.artifact.userId, userId), scopeCondition(ref), eq(schema.artifact.key, key)))
        .orderBy(desc(schema.artifact.version))
        .limit(1);
      return row ?? null;
    },
    async listKeys(ref: ArtifactScopeRef): Promise<string[]> {
      const rows = await db
        .selectDistinct({ key: schema.artifact.key })
        .from(schema.artifact)
        .where(and(eq(schema.artifact.userId, userId), scopeCondition(ref)));
      return rows.map((r) => r.key);
    },
    async insert(values: ArtifactFields): Promise<ArtifactRow> {
      const fields = stripScopedKeys(values);
      const [row] = await db
        .insert(schema.artifact)
        .values({ ...fields, userId })
        .returning();
      return row;
    },
    async update(id: string, values: Partial<ArtifactFields>): Promise<ArtifactRow | null> {
      const fields = stripScopedKeys(values);
      const [row] = await db
        .update(schema.artifact)
        .set({ ...fields, updatedAt: new Date() })
        .where(and(eq(schema.artifact.id, id), eq(schema.artifact.userId, userId)))
        .returning();
      return row ?? null;
    },
  };
}

`revoke`'s `coalesce(revoked_at, now)` needs the same explicit `::timestamptz` cast D4's rate-limit statement needs, for the same reason: PGlite cannot always infer a bind parameter's type from its position inside a SQL expression. ```typescript
import { and, desc, eq, sql } from "drizzle-orm";
import * as schema from "../schema";
import type { Db } from "../client";
import { stripScopedKeys } from "./strip";

export type ApiTokenRow = typeof schema.apiToken.$inferSelect;
export type ApiTokenListItem = Pick<ApiTokenRow, "id" | "name" | "prefix" | "createdAt" | "lastUsedAt" | "revokedAt">;

const LIST_COLUMNS = {
  id: schema.apiToken.id,
  name: schema.apiToken.name,
  prefix: schema.apiToken.prefix,
  createdAt: schema.apiToken.createdAt,
  lastUsedAt: schema.apiToken.lastUsedAt,
  revokedAt: schema.apiToken.revokedAt,
};

export function apiTokenQueries(db: Db, userId: string) {
  return {
    async list(): Promise<ApiTokenListItem[]> {
      return db
        .select(LIST_COLUMNS)
        .from(schema.apiToken)
        .where(eq(schema.apiToken.userId, userId))
        .orderBy(desc(schema.apiToken.createdAt));
    },
    async insert(values: { name: string; tokenHash: string; prefix: string }): Promise<ApiTokenListItem> {
      const fields = stripScopedKeys(values);
      const [row] = await db
        .insert(schema.apiToken)
        .values({ ...fields, userId })
        .returning(LIST_COLUMNS);
      return row;
    },
    async revoke(id: string, now: Date): Promise<ApiTokenListItem | null> {
      const [row] = await db
        .update(schema.apiToken)
        .set({ revokedAt: sql`coalesce(${schema.apiToken.revokedAt}, ${now.toISOString()}::timestamptz)` })
        .where(and(eq(schema.apiToken.id, id), eq(schema.apiToken.userId, userId)))
        .returning(LIST_COLUMNS);
      return row ?? null;
    },
  };
}
```

`LIST_COLUMNS` is used for `insert`'s own `.returning()` too, not only `list`, so `ApiTokenListItem`'s "never selects the hash" guarantee (D5) holds for every path that can hand a token row back to a caller other than `createApiToken` (Task 5), which is the one place the plain token string is ever produced, from a value it holds in memory, never from a re-read of `token_hash`.

- [ ] **Step 6: Implement `lib/artifacts/normalize.ts`**

`normalizeBody(raw: string): string` (D15)
1. If the first character's code point is `U+FEFF`, drop it (leading BOM).
2. Replace every `\r\n` with `\n`, then every remaining lone `\r` with `\n`.
3. Split on `\n`. For each line, strip a trailing run of spaces and/or tabs (a run at the very end of that line only; internal whitespace is untouched).
4. While the last element of the split array is the empty string, drop it (this removes every trailing blank line; because `"a\nb\n".split("\n")` ends in one empty element per trailing newline, dropping trailing empties also removes the final newline itself with no separate step).
5. Join the remaining lines with `\n` and return. Edge case: an input that is entirely whitespace (or empty) ends with every line stripped to `""`, so step 4 empties the whole array and this returns `""`.

`hashBody(normalized: string): string`
1. Return the sha256 hex digest of `normalized` (`node:crypto`'s `createHash("sha256").update(normalized, "utf8").digest("hex")`).

`utf8Bytes(value: string): number`
1. Return `Buffer.byteLength(value, "utf8")`.

`deriveTitle(input): string` (D16, `TITLE_MAX = 200`)
1. `given = input.title?.trim()`. If non-empty, return `given` clipped to 200 characters.
2. Otherwise scan `input.bodyMd`'s lines (split on `\r\n`, `\r` or `\n`, since this runs on the raw, not-yet-normalized body, see Task 4's `applyUpserts` step order) for the first ATX level-1 heading: a line matching "one `#`, not followed by another `#`, then one or more spaces, then the rest of the line" (a bare `##` is level 2 and never matches; a `#` with no following space is not a heading at all and never matches). Take the captured text, trim trailing whitespace, then strip a trailing run of `#` characters (with any whitespace right before them): this is the "closing hashes" an ATX heading is allowed to end with, for example `# Title ###` becomes `Title`. If found, return it clipped to 200 characters.
3. Otherwise return `input.key` clipped to 200 characters.

`keyFromTitle(title: string): string` (D17)
1. Run the exact same `slugify` steps `lib/pipeline/slug.ts`'s `baseSlug` already uses in this repo (NFKD-normalize, lowercase, strip everything that is not a Unicode letter, number, whitespace or hyphen, trim, collapse whitespace/underscore/hyphen runs into a single hyphen, strip leading/trailing hyphens), on `title` alone (no company/role concatenation).
2. Clip to 60 characters, then strip a trailing hyphen the clip may have introduced (same two-step shape `baseSlug` already uses for its own 60-character cap).
3. If the result is empty, return `"document"`; otherwise return it. The result always satisfies `KEY_PATTERN` (Task 3, `lib/bridge/wire.ts`): only lowercase letters, digits and hyphens ever survive step 1, leading/trailing hyphens are stripped in both steps 1 and 2, so a non-empty result always starts with a letter or digit and is at most 60 characters, well inside `KEY_PATTERN`'s 100-character ceiling.

`planUpsert(versions, intent, now): UpsertPlan`
1. Rule 1: if `versions` is empty, return `{ status: "created", insert: { version: 1, kind: intent.kind, title: intent.title, stageId: intent.stageId, bodyMd: intent.bodyMd, contentHash: intent.hash, sourceHash, origin: intent.origin, editedAt } }` where `sourceHash` is `null` when `intent.origin === "manual"`, else `intent.hash`; `editedAt` is `now` when `intent.origin === "manual"`, else `null`.
2. Let `latest = versions[versions.length - 1]` (the array is ascending by version, per this function's own contract).
3. If `intent.origin` is `"pushed"` or `"generated"` (rule 2): scan `versions` from the end backward for the first entry whose own `origin` is `"pushed"` or `"generated"`; call it `reference` (`null` if none exists; every prior version was pasted or manual). The body is unchanged when `intent.hash === reference?.sourceHash` OR `intent.hash === latest.contentHash`; when unchanged, go to step 6 (rule 6) with `latest` and `intent`. Otherwise return `{ status: "versioned", insert: { version: latest.version + 1, kind: intent.kind, title: intent.title, stageId: intent.stageId, bodyMd: intent.bodyMd, contentHash: intent.hash, sourceHash: intent.hash, origin: intent.origin, editedAt: null } }`.
4. Else if `intent.origin === "pasted"` (rule 3): the body is unchanged when `intent.hash === latest.contentHash`; when unchanged, go to step 6. Otherwise return the same `"versioned"` shape as step 3's else-branch (still `sourceHash: intent.hash`, `editedAt: null`), version `latest.version + 1`.
5. Else (`intent.origin === "manual"`, rule 4): when `intent.hash === latest.contentHash`, return `{ status: "unchanged", warning: null }` directly: do NOT go to step 6; the in-app editor never changes kind, title or stage (D14e), so a metadata patch here would always be empty by construction, and rule 4's own text states the "unchanged" outcome directly rather than deferring to rule 6. Otherwise, when `latest.sentAt` is set, return `{ status: "versioned", insert: { version: latest.version + 1, kind: latest.kind, title: latest.title, stageId: latest.stageId, bodyMd: intent.bodyMd, contentHash: intent.hash, sourceHash: null, origin: "manual", editedAt: now } }` (the metadata comes from `latest`, not `intent`). Otherwise return `{ status: "edited", id: latest.id, patch: { bodyMd: intent.bodyMd, contentHash: intent.hash, editedAt: now } }`.
6. (Rule 6, reached only from steps 3 and 4 when the body is unchanged.) Build `patch` from the fields among `kind`, `title`, `stageId` where `intent`'s value differs from `latest`'s own value; a field is only added to `patch` when it differs. If `patch` has no keys, return `{ status: "unchanged", warning: null }`. If `patch` has at least one key and `latest.sentAt` is set, return `{ status: "unchanged", warning: "sent_locked" }`. Otherwise return `{ status: "updated", id: latest.id, patch }`.

No step in this function ever sets `sentAt` on a new version: `NewVersion` has no such field, so "a new version never copies sentAt" (rule 5) falls out of the type, not a runtime check.

`versionNotice(versions, version): string | null`
1. Find the entry in `versions` whose own `version` field equals the `version` argument; if none, return `null`.
2. If that entry's `origin` is neither `"pushed"` nor `"generated"`, return `null`.
3. Find the entry whose `version` field equals `version - 1` (version numbers are contiguous within one key's history; `planUpsert` always inserts `latest.version + 1`, so "the previous version" is always exactly one less, never a gap); if none, or if it has no `editedAt`, return `null`.
4. Otherwise return the literal string `` `Pushed after you edited version ${previous.version}.` ``.

2. `company = docsForTab(companyDocs, "research")`. When non-empty, append `{ id: "company", heading: \`Shared with every job at ${companyName}\`, docs: company }`.
3. Return the groups that were appended (zero, one or both).

`prepGroups(docs, stages): DocGroup[]`
1. `prepDocs = docsForTab(docs, "prep")`.
2. `general = prepDocs` filtered to those whose `stageId` is `null` OR does not match any id in `stages` (an unknown stage id counts as General). When non-empty, this is the first group: `{ id: "general", heading: "General", docs: general }`.
3. Sort `stages` by `position` ascending. For each, `staged = prepDocs` filtered to `stageId === stage.id`; when non-empty, append `{ id: stage.id, heading: stage.label, docs: staged }`.
4. Return every group that was non-empty, in the order built (General, if present, always first).

`selectDoc(groups, ref): ArtifactMeta | null`
1. Flatten every group's `docs` into one list, in group order. If it is empty, return `null`.
2. When `ref` is given, look for a doc whose `scopeOf(doc) === ref.scope && doc.key === ref.key`; if found, return it.
3. Otherwise (no `ref`, or `ref` matched nothing) return the first doc in the flattened list.

`stagesWithArtifacts(jobDocs): Set<string>`
1. Return the set of every non-null `stageId` across `jobDocs`. (Callers must pass a latest-version-only list, per the Interfaces note above; this function has no way to enforce that itself.)

- [ ] **Step 18: Implement `lib/artifacts/upsert.ts`**

Imports: `Scoped`, `ArtifactScopeRef` (type) from `@/lib/db/scoped`; `Result`, `ok`, `fail` from `@/lib/result`; `parseKind`, `kindInfo`, `type ArtifactKind` from `@/lib/artifacts/kinds`; `matchStageRef` from `@/lib/artifacts/stage-ref`; `deriveTitle`, `normalizeBody`, `hashBody`, `utf8Bytes` from `@/lib/artifacts/normalize`; `planUpsert`, `type VersionState` from `@/lib/artifacts/plan`; `warningMessage` from `@/lib/artifacts/values`; `type ArtifactScope`, `type ArtifactWarning`, `type UpsertStatus` from `@/lib/artifacts/values`; `KEY_PATTERN`, `MAX_ARTIFACT_BYTES` from `@/lib/bridge/wire`.

`applyUpserts(tx, context, inputs, options): Promise<UpsertOutcome>`. The caller already holds the opportunity lock; this function locks the company row itself, once, the first time it is about to write a company-scoped artifact.

For each `input` in `inputs`, in this order:
1. `kind = parseKind(input.kind)`. If `null`: push a `{ key: input.key, code: "unknown_kind", message: warningMessage("unknown_kind", input.key, { kind: input.kind }) }` warning and use `kind = "other"` for the rest of this input.
2. `scope = input.scope`. If `scope === "company"` and `!kindInfo(kind).companyWide`: push a `company_scope_not_allowed` warning (detail: `{}`) and use `scope = "opportunity"` for the rest of this input.
3. `stageId = null`. If `input.stage` is set:
   - When `scope === "company"`: push a `stage_ignored` warning, detail `{ stage: "ref" in input.stage ? input.stage.ref : input.stage.id }`; `stageId` stays `null`.
   - Else when `input.stage` has a `ref`: `matched = matchStageRef(context.stages, input.stage.ref)`. If matched, `stageId = matched`; else push a `stage_not_found` warning, detail `{ stage: input.stage.ref }`.
   - Else (`input.stage` has an `id`): find it in `context.stages`. If found, `stageId` is that stage's own id; else push a `stage_not_found` warning, detail `{ stage: input.stage.id }` (an `{ id }` that names no real stage is treated the same as no match, per the frame: "`{ id }` must be one of `context.stages`").
4. `title = deriveTitle({ title: input.title, bodyMd: input.bodyMd, key: input.key })` (runs on the raw `input.bodyMd`, before normalization, per `deriveTitle`'s own H1-scanning contract).
5. `normalized = normalizeBody(input.bodyMd)`, `hash = hashBody(normalized)`.
6. `ref: ArtifactScopeRef` is `{ opportunityId: context.opportunity.id }` when `scope === "opportunity"`, else `{ companyId: context.opportunity.companyId }`.
7. `versions = await tx.artifact.listVersions(ref, input.key)`.
8. `plan = planUpsert(versions, { origin: options.origin, kind, title, stageId, bodyMd: normalized, hash }, options.now)`.
9. When `plan.status === "unchanged"` and `plan.warning === "sent_locked"`: push a `sent_locked` warning, detail `{ version: versions[versions.length - 1].version }` (the version that stayed locked).
10. Apply the plan, unless `options.dryRun`:
    - `"created"` or `"versioned"`: before the first such write when `scope === "company"`, `await tx.company.lockById(context.opportunity.companyId)`, once for the whole call (track this with a boolean flag declared before the loop). Then `await tx.artifact.insert({ opportunityId: scope === "opportunity" ? context.opportunity.id : null, companyId: scope === "company" ? context.opportunity.companyId : null, stageId: plan.insert.stageId, key: input.key, version: plan.insert.version, kind: plan.insert.kind, title: plan.insert.title, bodyMd: plan.insert.bodyMd, contentHash: plan.insert.contentHash, sourceHash: plan.insert.sourceHash, origin: plan.insert.origin, editedAt: plan.insert.editedAt })`. `version` for this input's result is `plan.insert.version` either way (dry run or not, since `planUpsert` already computed the number it would get).
    - `"updated"`: same company-lock check as above, then `await tx.artifact.update(versions[versions.length - 1].id, plan.patch)`. `version` is `versions[versions.length - 1].version`.
    - `"edited"`: same company-lock check, then `await tx.artifact.update(plan.id, plan.patch)`. `version` is `versions[versions.length - 1].version`.
    - `"unchanged"`: no write. `version` is `versions[versions.length - 1].version`.

7. `await tx.artifact.update(row.id, { sentAt: resolvedNow })`.
8. `await tx.event.insert({ opportunityId: opportunity.id, stageId: row.stageId, kind: "document_sent", occurredAt: resolvedNow, meta: { key, version, title: row.title } })`.
9. Return `ok({ title: row.title })`.
